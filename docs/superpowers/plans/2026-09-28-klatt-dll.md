**Goal:** Build the Klatt 1980 cascade/parallel formant synthesizer as `klatt32.dll` and `klatt64.dll` from one C
source, with a test suite that verifies it acoustically, and a Python `ctypes` sample.

**Architecture:** Plain C in `klatt/`: a header-only resonator (`resonator.h`), the parameter table
(`params.c`) and the synthesizer (`klatt.c`). The public API in `klatt.h` is frame-level: create a handle from
the global parameters, feed it one frame of parameters at a time, get 16-bit samples back. A console test runner
in `tests/` links the DLL through its import library, synthesizes known sounds and measures them back (F0 by
autocorrelation, formants by LPC).

**Tech Stack:** C (gnu11), winlibs GCC 14.2 (i686 and x86_64), GNU make 4.4 run from Git Bash, Python 3 for the
sample.

**Spec:** `docs/superpowers/specs/2026-09-28-kpe-open-design.md` (sections 2, 3, 4, 7, 8). Plan 2 (editor core)
and plan 3 (Win32 GUI) build on this one.

suite passed on both architectures (343 checks), and the 32- and 64-bit outputs were bit-identical. The
calibration constants in `klatt.c` are the measured values.

$(DLL): $(KLATT_SRC) $(KLATT_HDR) | $(OUT)

else
$(DLL): $(KLATT_SRC) $(KLATT_HDR) | $(OUT)
	$(CC) $(CFLAGS) -DKLATT_BUILD -shared -o $@ $(KLATT_SRC) -static-libgcc -Wl,--out-implib,$(IMPLIB)
endif

$(TESTS): $(TEST_SRC) $(wildcard tests/*.h) $(DLL) | $(OUT)
	$(CC) $(CFLAGS) -Itests -o $@ $(TEST_SRC) $(IMPLIB) -static

all:
	$(MAKE) ARCH=32 build

/* klatt.h - public API of the Klatt 1980 cascade/parallel formant synthesizer DLL.
 *
 * Built as klatt32.dll (i686) and klatt64.dll (x86_64). On 32-bit Windows every function is __stdcall and
 * exported under its plain name (KlattCreate, not _KlattCreate@8). Functions returning int return 0 or more
 * on success and a negative KLATT_ERR_* code on failure. Parameter values are doubles in physical units. */
#ifndef KLATT_H
#define KLATT_H

#ifdef _WIN32
#  define KLATT_CALL __stdcall
#  ifdef KLATT_BUILD
#    define KLATT_API __declspec(dllexport)
#  else
#    define KLATT_API __declspec(dllimport)
#  endif
#else
#  define KLATT_CALL
#  define KLATT_API
#endif

#ifdef __cplusplus
extern "C" {
#endif

#define KLATT_VERSION 10000 /* 1.0.0, as major*10000 + minor*100 + patch */

enum { KLATT_FRAME = 0, KLATT_GLOBAL = 1 }; /* parameter kinds */

enum {
	KLATT_OK = 0,
	KLATT_ERR_ARG = -1,    /* NULL pointer, bad kind or index */
	KLATT_ERR_BUFFER = -2, /* output buffer smaller than the frame */
	KLATT_ERR_MEMORY = -3
};

/* Frame parameters, in the order KlattSynthFrame expects them. */
enum {
	KP_AV, KP_AF, KP_AH, KP_AVS, KP_F0, KP_F1, KP_F2, KP_F3, KP_F4, KP_FNZ, KP_AN,
	KP_A1, KP_A2, KP_A3, KP_A4, KP_A5, KP_A6, KP_AB, KP_B1, KP_B2, KP_B3,
	KLATT_NUM_FRAME
};

/* Global parameters, in the order KlattCreate expects them. */
enum {
	KG_SW, KG_FGP, KG_BGP, KG_FGZ, KG_BGZ, KG_BGS, KG_FNP, KG_BNP, KG_BNZ,
	KG_B4, KG_F5, KG_B5, KG_F6, KG_B6, KG_SR, KG_FRAME_MS, KG_G0, KG_NFC,
	KLATT_NUM_GLOBAL
};

typedef struct klatt_synth *klatt_h;

typedef struct {
	const char *name, *unit, *description;
	double min, max, def;
} klatt_param_info;

KLATT_API int KLATT_CALL KlattVersion(void);
KLATT_API int KLATT_CALL KlattParamCount(int kind);
KLATT_API int KLATT_CALL KlattParamInfo(int kind, int index, klatt_param_info *out);
KLATT_API int KLATT_CALL KlattParamIndex(int kind, const char *name);
KLATT_API const char *KLATT_CALL KlattErrorText(int code);

KLATT_API int KLATT_CALL KlattCreate(const double *globals, klatt_h *out);
KLATT_API int KLATT_CALL KlattMaxFrameSamples(klatt_h h);
KLATT_API int KLATT_CALL KlattSynthFrame(klatt_h h, const double *frame, short *out, int max);
KLATT_API int KLATT_CALL KlattOverloadCount(klatt_h h);
KLATT_API void KLATT_CALL KlattReset(klatt_h h, unsigned seed);
KLATT_API void KLATT_CALL KlattDestroy(klatt_h h);

#ifdef __cplusplus
}
#endif
#endif
```

/* test.h - minimal test harness: CHECK macros and suite registration. */
#ifndef TEST_H
#define TEST_H

#include <math.h>
#include <stdio.h>

extern int t_checks, t_fails;

#define CHECK(c) do { \
	t_checks++; \
	if (!(c)) { t_fails++; printf("  FAIL %s:%d: %s\n", __FILE__, __LINE__, #c); } \
} while (0)

#define CHECK_NEAR(a, b, tol) do { \
	double a_ = (a), b_ = (b), t_ = (tol); \
	t_checks++; \
	if (!(fabs(a_ - b_) <= t_)) { \
		t_fails++; \
		printf("  FAIL %s:%d: %s = %g, expected %g +- %g\n", __FILE__, __LINE__, #a, a_, b_, t_); \
	} \
} while (0)

#endif
```

/* run_tests.c - runs every suite; exit code 0 when all checks pass. */
#include <stdio.h>
#include "test.h"

int t_checks, t_fails;

void suite_params(void);

static const struct { const char *name; void (*fn)(void); } suites[] = {
	{ "params", suite_params },
};

int main(void)
{
	for (size_t i = 0; i < sizeof suites / sizeof suites[0]; i++) {
		int before = t_fails;
		printf("%s\n", suites[i].name);
		suites[i].fn();
		printf("  %s\n", t_fails == before ? "ok" : "FAILED");
	}
	printf("%d checks, %d failures\n", t_checks, t_fails);
	return t_fails ? 1 : 0;
}

static const klatt_param_info *table(int kind, int *count)
{
	if (kind == KLATT_FRAME) { *count = KLATT_NUM_FRAME; return frame_params; }
	if (kind == KLATT_GLOBAL) { *count = KLATT_NUM_GLOBAL; return global_params; }
	*count = 0;
	return NULL;
}

double klatt_clamp(int kind, int index, double v)
{
	int n;
	const klatt_param_info *t = table(kind, &n);
	if (!t || index < 0 || index >= n)
		return v;
	if (!(v >= t[index].min)) /* also catches NaN */
		return t[index].min;
	if (v > t[index].max)
		return t[index].max;
	return v;
}

KLATT_API int KLATT_CALL KlattVersion(void)
{
	return KLATT_VERSION;
}

KLATT_API int KLATT_CALL KlattParamCount(int kind)
{
	int n;
	return table(kind, &n) ? n : KLATT_ERR_ARG;
}

KLATT_API int KLATT_CALL KlattParamInfo(int kind, int index, klatt_param_info *out)
{
	int n;
	const klatt_param_info *t = table(kind, &n);
	if (!t || !out || index < 0 || index >= n)
		return KLATT_ERR_ARG;
	*out = t[index];
	return KLATT_OK;
}

KLATT_API int KLATT_CALL KlattParamIndex(int kind, const char *name)
{
	int n;
	const klatt_param_info *t = table(kind, &n);
	if (!t || !name)
		return KLATT_ERR_ARG;
	for (int i = 0; i < n; i++) {
		const char *a = t[i].name, *b = name;
		while (*a && toupper((unsigned char)*a) == toupper((unsigned char)*b))
			a++, b++;
		if (!*a && !*b)
			return i;
	}
	return KLATT_ERR_ARG;
}

KLATT_API const char *KLATT_CALL KlattErrorText(int code)
{
	switch (code) {
	case KLATT_OK:         return "no error";
	case KLATT_ERR_ARG:    return "invalid argument";
	case KLATT_ERR_BUFFER: return "output buffer too small for the frame";
	case KLATT_ERR_MEMORY: return "out of memory";
	default:               return "unknown error";
	}
}

#include "test.h"
#include "resonator.h"

#define SR 11025.0

/* Peak frequency and -3 dB width of r, searched on a 0.1 Hz grid. */
static void peak_and_width(const reson *r, double *peak_f, double *width)
{
	double best = 0, bf = 0;
	for (double f = 1; f < SR / 2; f += 0.1) {
		double g = reson_gain(r, f, SR);
		if (g > best) { best = g; bf = f; }
	}
	double half = best / sqrt(2.0), lo = bf, hi = bf;
	while (lo > 0.1 && reson_gain(r, lo, SR) > half) lo -= 0.1;
	while (hi < SR / 2 && reson_gain(r, hi, SR) > half) hi += 0.1;
	*peak_f = bf;
	*width = hi - lo;
}

static void test_resonator_peak_bandwidth_dc(void)
{
	static const double cases[][2] = { { 500, 60 }, { 1500, 90 }, { 2500, 150 }, { 3500, 250 } };
	for (int i = 0; i < 4; i++) {
		reson r = { 0 };
		reson_set(&r, cases[i][0], cases[i][1], SR);
		double pf, w;
		peak_and_width(&r, &pf, &w);
		CHECK_NEAR(pf, cases[i][0], cases[i][0] * 0.01);
		CHECK_NEAR(w, cases[i][1], cases[i][1] * 0.02);
		CHECK_NEAR(reson_gain(&r, 0, SR), 1.0, 1e-9);
	}
}

static void test_resonator_step_matches_response(void)
{
	/* A long sine through the resonator settles to the analytic gain. */
	reson r = { 0 };
	reson_set(&r, 1000, 100, SR);
	double peak = 0;
	for (int n = 0; n < 20000; n++) {
		double y = reson_step(&r, sin(2 * KLATT_PI * 1000 * n / SR));
		if (n > 10000 && fabs(y) > peak) peak = fabs(y);
	}
	CHECK_NEAR(peak, reson_gain(&r, 1000, SR), 0.01 * reson_gain(&r, 1000, SR));
}

static void test_antiresonator_notch_and_dc(void)
{
	antireson z = { 0 };
	antireson_set(&z, 400, 100, SR);
	double best = 1e9, bf = 0;
	for (double f = 1; f < SR / 2; f += 0.1) {
		double g = antireson_gain(&z, f, SR);
		if (g < best) { best = g; bf = f; }
	}
	CHECK_NEAR(bf, 400, 4);

```c
/* resonator.h - Klatt (1980) digital resonator and antiresonator.
 *
 * Resonator:      y[n] = A*x[n] + B*y[n-1] + C*y[n-2]
 *                 C = -exp(-2*pi*BW*T), B = 2*exp(-pi*BW*T)*cos(2*pi*F*T), A = 1 - B - C (unity gain at DC)
 * Antiresonator:  y[n] = A'*x[n] + B'*x[n-1] + C'*x[n-2], with A' = 1/A, B' = -B/A, C' = -C/A
 *                 (the inverse of the resonator with the same F and BW). */
#ifndef KLATT_RESONATOR_H
#define KLATT_RESONATOR_H

#include <math.h>

#define KLATT_PI 3.14159265358979323846

typedef struct { double a, b, c, y1, y2; } reson;
typedef struct { double a, b, c, x1, x2; } antireson;

/* Sets the coefficients; keeps the filter state so parameters can change between frames. */
static inline void reson_set(reson *r, double f, double bw, double sr)
{
	double t = 1.0 / sr;
	r->c = -exp(-2.0 * KLATT_PI * bw * t);
	r->b = 2.0 * exp(-KLATT_PI * bw * t) * cos(2.0 * KLATT_PI * f * t);
	r->a = 1.0 - r->b - r->c;
}

static inline double reson_step(reson *r, double x)
{
	double y = r->a * x + r->b * r->y1 + r->c * r->y2;
	r->y2 = r->y1;
	r->y1 = y;
	return y;
}

static inline void antireson_set(antireson *r, double f, double bw, double sr)
{
	reson t;
	reson_set(&t, f, bw, sr);
	r->a = 1.0 / t.a;
	r->b = -t.b / t.a;
	r->c = -t.c / t.a;
}

static inline double antireson_step(antireson *r, double x)
{
	double y = r->a * x + r->b * r->x1 + r->c * r->x2;
	r->x2 = r->x1;
	r->x1 = x;
	return y;
}

/* Magnitude responses at frequency f, for tests: |A / (1 - B z^-1 - C z^-2)| and |A + B z^-1 + C z^-2|. */
static inline double reson_gain(const reson *r, double f, double sr)
{
	double w = 2.0 * KLATT_PI * f / sr;
	double re = 1.0 - r->b * cos(w) - r->c * cos(2.0 * w);
	double im = r->b * sin(w) + r->c * sin(2.0 * w);
	return fabs(r->a) / sqrt(re * re + im * im);
}

static inline double antireson_gain(const antireson *r, double f, double sr)
{
	double w = 2.0 * KLATT_PI * f / sr;
	double re = r->a + r->b * cos(w) + r->c * cos(2.0 * w);
	double im = -r->b * sin(w) - r->c * sin(2.0 * w);
	return sqrt(re * re + im * im);
}

#endif
```

### Task 3: The synthesizer

The signal path follows the spec's section 4.1. In brief, per sample:

1. **Voicing:** an impulse at the start of each glottal period goes through RGP. The result goes through RGZ,
   scaled by AV, plus through RGS, scaled by AVS. AV, AVS and the period length change only at a period start.
2. **Noise:** four uniform random numbers are summed and low-pass filtered (`y = x + 0.9·y[n-1]`), and halved in
   the second half of each voiced period. AH scales it for aspiration and AF for frication.
3. **Cascade** (SW = 0): voicing + aspiration → nasal pole → nasal zero → formants R`NFC` … R1.
4. **Parallel:** R1 gets the undifferenced voicing (SW = 1 only). R2–R6 get the first difference of the voicing
   (SW = 1 only) plus frication. The nasal pole gets the voicing difference, and the bypass gets the R2 input.
   Signs alternate starting with + for R1.
5. **Output:** first difference, then × `K_OUT·10^(G0/20)`, then round and saturate to 16 bits, counting
   saturations.

Amplitudes in dB become linear gain as `0` for 0 dB and `10^((dB − 80)/20)` above it. Three calibration
constants set the levels. They were measured with `tests/calibrate.c` (Step 6):

- `K_OUT = 4.125e5`: a neutral vowel with AV = 60 and G0 = 47 peaks near 16000.
- `K_NOISE = 4.071e-4`: AH = 60 alone has the same RMS as AV = 60 alone.
- `K_PAR = 6.0`: an all-parallel vowel with A1–A4 = 60 has the same RMS as the cascade vowel.

Frame *k* ends at sample `floor((k+1)·SR·FRAME_MS/1000)`, so 5 ms frames at 11025 Hz are 55 or 56 samples.

#include <math.h>
#include <stdlib.h>
#include "klatt.h"
#include "measure.h"

#define PI 3.14159265358979323846

void t_default_globals(double *g)
{
	for (int i = 0; i < KLATT_NUM_GLOBAL; i++) {
		klatt_param_info pi;
		KlattParamInfo(KLATT_GLOBAL, i, &pi);
		g[i] = pi.def;
	}
}

void t_default_frame(double *f)
{
	for (int i = 0; i < KLATT_NUM_FRAME; i++) {
		klatt_param_info pi;
		KlattParamInfo(KLATT_FRAME, i, &pi);
		f[i] = pi.def;
	}
}

int t_synth_steady(const double *globals, const double *frame, int nframes, short *out, int max)
{
	klatt_h h;
	int r = KlattCreate(globals, &h);
	if (r < 0)
		return r;
	int total = 0;
	for (int k = 0; k < nframes; k++) {
		r = KlattSynthFrame(h, frame, out + total, max - total);
		if (r < 0) {
			KlattDestroy(h);
			return r;
		}
		total += r;
	}
	KlattDestroy(h);
	return total;
}

double m_rms(const short *x, int n)
{
	double s = 0;
	for (int i = 0; i < n; i++)
		s += (double)x[i] * x[i];
	return n ? sqrt(s / n) : 0;
}

double m_peak(const short *x, int n)
{
	double p = 0;
	for (int i = 0; i < n; i++)
		if (fabs((double)x[i]) > p)
			p = fabs((double)x[i]);
	return p;
}

double m_f0(const short *x, int n, double sr, double fmin, double fmax)
{
	int lo = (int)(sr / fmax), hi = (int)(sr / fmin) + 1;
	double best = -2, r[3] = { 0 };
	int bl = lo;
	for (int lag = lo; lag <= hi + 1; lag++) {
		double xy = 0, xx = 0, yy = 0;
		for (int i = 0; i + lag < n; i++) {
			xy += (double)x[i] * x[i + lag];
			xx += (double)x[i] * x[i];
			yy += (double)x[i + lag] * x[i + lag];
		}
		double c = xy / sqrt(xx * yy + 1e-12);
		if (lag <= hi && c > best) { best = c; bl = lag; }
	}
	/* parabolic refinement around the best lag */
	for (int k = -1; k <= 1; k++) {
		double xy = 0, xx = 0, yy = 0;
		int lag = bl + k;
		for (int i = 0; i + lag < n; i++) {
			xy += (double)x[i] * x[i + lag];
			xx += (double)x[i] * x[i];
			yy += (double)x[i + lag] * x[i + lag];
		}
		r[k + 1] = xy / sqrt(xx * yy + 1e-12);
	}
	double d = r[0] - 2 * r[1] + r[2];
	double lag = bl + (d != 0 ? 0.5 * (r[0] - r[2]) / d : 0);
	return sr / lag;
}

int m_formants(const short *x, int n, double sr, double *f, int maxf)
{
	enum { N = 1024, P = 12 };
	if (n < N + 2)
		return 0;
	int start = (n - N) / 2 + 1; /* leaves x[start - 1] for the pre-emphasis */
	double w[N], r[P + 1], a[P + 1] = { 1 }, tmp[P + 1];
	for (int i = 0; i < N; i++) {
		double pre = x[start + i] - 0.9 * x[start + i - 1];
		w[i] = pre * (0.54 - 0.46 * cos(2 * PI * i / (N - 1)));
	}
	for (int k = 0; k <= P; k++) {
		r[k] = 0;
		for (int i = k; i < N; i++)
			r[k] += w[i] * w[i - k];
	}
	double e = r[0];
	for (int i = 1; i <= P; i++) { /* Levinson-Durbin */
		double acc = r[i];
		for (int j = 1; j < i; j++)
			acc += a[j] * r[i - j];
		double k = -acc / e;
		for (int j = 0; j <= i; j++)
			tmp[j] = a[j];
		for (int j = 1; j < i; j++)
			a[j] = tmp[j] + k * tmp[i - j];
		a[i] = k;
		e *= 1 - k * k;
	}
	int count = 0;
	double prev2 = 0, prev1 = 0;
	for (double fr = 50; fr < sr / 2 - 50 && count < maxf; fr += 2) {
		double re = 0, im = 0, wv = 2 * PI * fr / sr;
		for (int j = 0; j <= P; j++) {
			re += a[j] * cos(wv * j);
			im -= a[j] * sin(wv * j);
		}
		double env = 1 / sqrt(re * re + im * im);
		if (fr > 54 && prev1 > prev2 && prev1 > env)
			f[count++] = fr - 2;
		prev2 = prev1;
		prev1 = env;
	}
	return count;
}

double m_power_at(const short *x, int n, double sr, double f)
{
	double w = 2 * PI * f / sr, re = 0, im = 0;
	for (int i = 0; i < n; i++) {
		re += x[i] * cos(w * i);
		im -= x[i] * sin(w * i);
	}
	return 2 * (re * re + im * im) / ((double)n * n);
}

double m_band_fraction(const short *x, int n, double sr, double lo, double hi)
{
	int N = n < 2048 ? n : 2048;
	double in = 0, total = 0;
	for (int k = 1; k < N / 2; k++) {
		double re = 0, im = 0, w = 2 * PI * k / N;
		for (int i = 0; i < N; i++) {
			double h = 0.5 - 0.5 * cos(2 * PI * i / (N - 1));
			re += h * x[i] * cos(w * i);
			im -= h * x[i] * sin(w * i);
		}
		double p = re * re + im * im, fr = k * sr / N;
		total += p;
		if (fr >= lo && fr <= hi)
			in += p;
	}
	return total > 0 ? in / total : 0;
}

/* klatt.c - Klatt (1980) cascade/parallel formant synthesizer, implemented from the paper's equations and
 * block diagram (D.H. Klatt, JASA 67, 971-995). See docs/klatt-dll.md for the signal path and calibration. */
#include <math.h>
#include <stdint.h>
#include <stdlib.h>
#include "klatt.h"
#include "klatt_internal.h"
#include "resonator.h"

/* Calibration constants, measured as described in docs/klatt-dll.md. */
#ifndef K_OUT
#define K_OUT   4.125e5 /* neutral vowel, AV = 60, G0 = 47: peak about 16000 */
#endif
#ifndef K_NOISE
#define K_NOISE 4.071e-4 /* AH = 60 alone: same RMS as AV = 60 alone */
#endif
#ifndef K_PAR
#define K_PAR   6.0 /* SW = 1, A1-A4 = 60: same RMS as the cascade vowel */
#endif

#define NOISE_POLE 0.9 /* noise low-pass y[n] = x[n] + NOISE_POLE*y[n-1]; with the radiation difference it is flat */

struct klatt_synth {
	double g[KLATT_NUM_GLOBAL]; /* clamped globals */
	double sr, spf, fmax;       /* sample rate, samples per frame, highest allowed frequency */
	int nfc, parallel_only;
	double out_gain;
	double frame;               /* frames synthesized since the last reset */
	uint32_t rng;
	int overload;

	/* voicing source; av/avs/period change only at the start of a glottal period */
	double period_len, period_pos;
	double av, avs, av_next, avs_next, f0_next;
	reson rgp, rgs;
	antireson rgz;

	double noise_lp;
	double ah, af;              /* linear noise gains of the current frame */

	reson rnp;                  /* cascade nasal pole */
	antireson rnz;              /* cascade nasal zero */
	reson rc[6];                /* cascade formants F1..F6 */

	reson rp[6];                /* parallel formants F1..F6 */
	reson rnp_par;              /* parallel nasal formant */
	double a[6], an, ab;        /* linear parallel gains of the current frame */
	double par_prev;            /* previous parallel voicing input, for its first difference */
	double rad_prev;            /* previous output, for the radiation first difference */
};

static double db_to_lin(double db)
{
	return db <= 0.0 ? 0.0 : pow(10.0, (db - 80.0) / 20.0);
}

static double noise(struct klatt_synth *s)
{
	double sum = 0.0;
	for (int i = 0; i < 4; i++) {
		s->rng = s->rng * 1664525u + 1013904223u;
		sum += (double)(s->rng >> 8) / 16777216.0;
	}
	return sum - 2.0; /* approximately Gaussian, mean 0 */
}

static void clear_reson(reson *r) { r->y1 = r->y2 = 0.0; }
static void clear_anti(antireson *r) { r->x1 = r->x2 = 0.0; }

KLATT_API int KLATT_CALL KlattCreate(const double *globals, klatt_h *out)
{
	if (!globals || !out)
		return KLATT_ERR_ARG;
	struct klatt_synth *s = calloc(1, sizeof *s);
	if (!s)
		return KLATT_ERR_MEMORY;
	for (int i = 0; i < KLATT_NUM_GLOBAL; i++)
		s->g[i] = klatt_clamp(KLATT_GLOBAL, i, globals[i]);
	s->sr = s->g[KG_SR];
	s->spf = s->sr * s->g[KG_FRAME_MS] / 1000.0;
	s->fmax = s->sr / 2.0 - 1.0;
	s->nfc = (int)lround(s->g[KG_NFC]);
	s->parallel_only = s->g[KG_SW] >= 0.5;
	s->out_gain = K_OUT * pow(10.0, s->g[KG_G0] / 20.0);

	reson_set(&s->rgp, s->g[KG_FGP], s->g[KG_BGP], s->sr);
	antireson_set(&s->rgz, fmin(s->g[KG_FGZ], s->fmax), s->g[KG_BGZ], s->sr);
	reson_set(&s->rgs, s->g[KG_FGP], s->g[KG_BGS], s->sr);
	reson_set(&s->rnp, s->g[KG_FNP], s->g[KG_BNP], s->sr);
	reson_set(&s->rnp_par, s->g[KG_FNP], s->g[KG_BNP], s->sr);
	reson_set(&s->rc[4], fmin(s->g[KG_F5], s->fmax), s->g[KG_B5], s->sr);
	reson_set(&s->rc[5], fmin(s->g[KG_F6], s->fmax), s->g[KG_B6], s->sr);
	s->rp[4] = s->rc[4];
	s->rp[5] = s->rc[5];
	KlattReset(s, 1);
	*out = s;
	return KLATT_OK;
}

KLATT_API void KLATT_CALL KlattReset(klatt_h s, unsigned seed)
{
	if (!s)
		return;
	s->frame = 0.0;
	s->rng = seed;
	s->overload = 0;
	s->period_len = s->period_pos = 0.0;
	s->av = s->avs = s->av_next = s->avs_next = s->f0_next = 0.0;
	s->noise_lp = s->par_prev = s->rad_prev = 0.0;
	clear_reson(&s->rgp);
	clear_reson(&s->rgs);
	clear_anti(&s->rgz);
	clear_reson(&s->rnp);
	clear_anti(&s->rnz);
	clear_reson(&s->rnp_par);
	for (int k = 0; k < 6; k++) {
		clear_reson(&s->rc[k]);
		clear_reson(&s->rp[k]);
	}
}

KLATT_API void KLATT_CALL KlattDestroy(klatt_h s)
{
	free(s);
}

KLATT_API int KLATT_CALL KlattMaxFrameSamples(klatt_h s)
{
	return s ? (int)ceil(s->spf) : KLATT_ERR_ARG;
}

KLATT_API int KLATT_CALL KlattOverloadCount(klatt_h s)
{
	return s ? s->overload : KLATT_ERR_ARG;
}

/* Loads one frame's parameters into the filters and gains. */
static void load_frame(struct klatt_synth *s, const double *in)
{
	double p[KLATT_NUM_FRAME];
	for (int i = 0; i < KLATT_NUM_FRAME; i++)
		p[i] = klatt_clamp(KLATT_FRAME, i, in[i]);

	static const int fi[4] = { KP_F1, KP_F2, KP_F3, KP_F4 };
	double bw[4] = { p[KP_B1], p[KP_B2], p[KP_B3], s->g[KG_B4] };
	for (int k = 0; k < 4; k++) {
		reson_set(&s->rc[k], fmin(p[fi[k]], s->fmax), bw[k], s->sr);
		reson_set(&s->rp[k], fmin(p[fi[k]], s->fmax), bw[k], s->sr);
	}
	antireson_set(&s->rnz, fmin(p[KP_FNZ], s->fmax), s->g[KG_BNZ], s->sr);

	s->av_next = db_to_lin(p[KP_AV]);
	s->avs_next = db_to_lin(p[KP_AVS]);
	s->f0_next = p[KP_F0];
	s->ah = K_NOISE * db_to_lin(p[KP_AH]);
	s->af = K_NOISE * db_to_lin(p[KP_AF]);
	for (int k = 0; k < 6; k++)
		s->a[k] = K_PAR * db_to_lin(p[KP_A1 + k]);
	s->an = K_PAR * db_to_lin(p[KP_AN]);
	s->ab = K_PAR * db_to_lin(p[KP_AB]);
}

static double sample(struct klatt_synth *s)
{
	/* Voicing: an impulse at the start of each glottal period. */
	double pulse = 0.0;
	if (s->period_len <= 0.0 || s->period_pos >= s->period_len) {
		s->period_pos = s->period_len > 0.0 ? s->period_pos - s->period_len : 0.0;
		s->av = s->av_next;
		s->avs = s->avs_next;
		s->period_len = s->f0_next > 0.0 ? s->sr / s->f0_next : 0.0;
		if (s->period_len > 0.0)
			pulse = 1.0;
	}
	if (s->period_len > 0.0)
		s->period_pos += 1.0;
	double gp = reson_step(&s->rgp, pulse);
	double voice = s->av * antireson_step(&s->rgz, gp) + s->avs * reson_step(&s->rgs, gp);

	/* Noise, halved in the second half of each voiced period. */
	s->noise_lp = noise(s) + NOISE_POLE * s->noise_lp;
	double n = s->noise_lp;
	if (s->period_len > 0.0 && s->av > 0.0 && s->period_pos > 0.5 * s->period_len)
		n *= 0.5;
	double asp = s->ah * n, fric = s->af * n;

	/* Cascade branch: voicing and aspiration (unless SW = 1). */
	double casc = 0.0, par_voice = 0.0;
	if (s->parallel_only) {
		par_voice = voice + asp;
	} else {
		double x = reson_step(&s->rnp, voice + asp);
		x = antireson_step(&s->rnz, x);
		for (int k = s->nfc - 1; k >= 0; k--)
			x = reson_step(&s->rc[k], x);
		casc = x;
	}

	/* Parallel branch: frication, plus voicing when SW = 1. F1 gets undifferenced voicing only;
	 * the others get the first difference of the voicing plus frication. Signs alternate. */
	double dv = par_voice - s->par_prev;
	s->par_prev = par_voice;
	double src = dv + fric;
	double par = s->a[0] * reson_step(&s->rp[0], par_voice)
	           + s->an * reson_step(&s->rnp_par, dv)
	           - s->a[1] * reson_step(&s->rp[1], src)
	           + s->a[2] * reson_step(&s->rp[2], src)
	           - s->a[3] * reson_step(&s->rp[3], src)
	           + s->a[4] * reson_step(&s->rp[4], src)
	           - s->a[5] * reson_step(&s->rp[5], src)
	           + s->ab * src;

	/* Radiation: first difference. */
	double sum = casc + par;
	double out = sum - s->rad_prev;
	s->rad_prev = sum;
	return out * s->out_gain;
}

KLATT_API int KLATT_CALL KlattSynthFrame(klatt_h s, const double *frame, short *out, int max)
{
	if (!s || !frame || !out)
		return KLATT_ERR_ARG;
	int n = (int)(floor((s->frame + 1.0) * s->spf + 1e-9) - floor(s->frame * s->spf + 1e-9));
	if (max < n)
		return KLATT_ERR_BUFFER;
	load_frame(s, frame);
	for (int i = 0; i < n; i++) {
		double v = floor(sample(s) + 0.5);
		if (v > 32767.0) { v = 32767.0; s->overload++; }
		if (v < -32768.0) { v = -32768.0; s->overload++; }
		out[i] = (short)v;
	}
	s->frame += 1.0;
	return n;
}

/* xarch_cmp.c - compares the xarch.raw files written by the 32- and 64-bit test runs; they must have the same
 * length and differ by at most 1 in any sample. Usage: xarch_cmp a.raw b.raw */
#include <stdio.h>
#include <stdlib.h>

static short *load(const char *path, long *n)
{
	FILE *f = fopen(path, "rb");
	if (!f)
		return NULL;
	fseek(f, 0, SEEK_END);
	long bytes = ftell(f);
	fseek(f, 0, SEEK_SET);
	short *p = malloc(bytes > 0 ? bytes : 1);
	*n = (long)fread(p, sizeof(short), bytes / sizeof(short), f);
	fclose(f);
	return p;
}

int main(int argc, char **argv)
{
	long na, nb;
	if (argc != 3) {
		fprintf(stderr, "usage: xarch_cmp a.raw b.raw\n");
		return 2;
	}
	short *a = load(argv[1], &na), *b = load(argv[2], &nb);
	if (!a || !b) {
		fprintf(stderr, "xarch_cmp: cannot read %s\n", !a ? argv[1] : argv[2]);
		return 2;
	}
	if (na != nb || na == 0) {
		printf("xarch: FAILED, lengths %ld and %ld\n", na, nb);
		return 1;
	}
	int worst = 0;
	for (long i = 0; i < na; i++) {
		int d = abs(a[i] - b[i]);
		if (d > worst)
			worst = d;
	}
	printf("xarch: %ld samples, max difference %d LSB: %s\n", na, worst, worst <= 1 ? "ok" : "FAILED");
	return worst <= 1 ? 0 : 1;
}

from collections import namedtuple

KLATT_FRAME, KLATT_GLOBAL = 0, 1
Param = namedtuple("Param", "name unit description min max default")


class ParamInfo(ctypes.Structure):
    _fields_ = [("name", ctypes.c_char_p), ("unit", ctypes.c_char_p), ("description", ctypes.c_char_p),
                ("min", ctypes.c_double), ("max", ctypes.c_double), ("default", ctypes.c_double)]


class Klatt:
    """One synthesizer handle. Frame and global parameters are dicts keyed by name; missing ones use defaults."""

    def __init__(self, dll_path, **globals_):
        self.lib = lib = ctypes.WinDLL(dll_path)  # __stdcall on 32-bit; the only convention on 64-bit
        lib.KlattParamCount.argtypes = [ctypes.c_int]
        lib.KlattParamInfo.argtypes = [ctypes.c_int, ctypes.c_int, ctypes.POINTER(ParamInfo)]
        lib.KlattErrorText.restype = ctypes.c_char_p
        lib.KlattCreate.argtypes = [ctypes.POINTER(ctypes.c_double), ctypes.POINTER(ctypes.c_void_p)]
        lib.KlattMaxFrameSamples.argtypes = [ctypes.c_void_p]
        lib.KlattSynthFrame.argtypes = [ctypes.c_void_p, ctypes.POINTER(ctypes.c_double),
                                        ctypes.POINTER(ctypes.c_short), ctypes.c_int]
        lib.KlattDestroy.argtypes = [ctypes.c_void_p]
        self.frame_params = self._table(KLATT_FRAME)
        self.global_params = self._table(KLATT_GLOBAL)
        g = self._values(self.global_params, globals_)
        self.sr = g[[p.name for p in self.global_params].index("SR")]
        self.handle = ctypes.c_void_p()
        self._check(lib.KlattCreate(g, ctypes.byref(self.handle)))
        self.buf = (ctypes.c_short * lib.KlattMaxFrameSamples(self.handle))()

    def _table(self, kind):
        out = []
        for i in range(self.lib.KlattParamCount(kind)):
            info = ParamInfo()
            self._check(self.lib.KlattParamInfo(kind, i, ctypes.byref(info)))
            out.append(Param(info.name.decode(), info.unit.decode(), info.description.decode(),
                             info.min, info.max, info.default))
        return out

    def _values(self, table, given):
        unknown = set(given) - {p.name for p in table}
        if unknown:
            raise KeyError(f"unknown parameters: {sorted(unknown)}")
        return (ctypes.c_double * len(table))(*[given.get(p.name, p.default) for p in table])

    def _check(self, code):
        if code < 0:
            raise RuntimeError(self.lib.KlattErrorText(code).decode())
        return code

    def frame(self, **params):
        """Synthesizes one frame and returns its samples as a list of ints."""
        n = self._check(self.lib.KlattSynthFrame(self.handle, self._values(self.frame_params, params),
                                                  self.buf, len(self.buf)))
        return self.buf[:n]

    def close(self):
        if self.handle:
            self.lib.KlattDestroy(self.handle)
            self.handle = None


def main():
    dll, out = sys.argv[1], sys.argv[2]
    k = Klatt(dll)
    samples = []
    frames = 100  # 100 x 5 ms = 0.5 s: "ah" gliding to "ee"
    for i in range(frames):
        t = i / (frames - 1)
        samples += k.frame(AV=60, F0=130 - 30 * t,
                           F1=700 + (300 - 700) * t, F2=1220 + (2250 - 1220) * t, F3=2600 + (3000 - 2600) * t)
    k.close()
    with wave.open(out, "wb") as w:
        w.setnchannels(1)
        w.setsampwidth(2)
        w.setframerate(int(k.sr))
        w.writeframes(struct.pack(f"<{len(samples)}h", *samples))
    print(f"wrote {len(samples)} samples to {out}")


if __name__ == "__main__":
    main()

# Klatt synthesizer DLL

`klatt32.dll` (32-bit) and `klatt64.dll` (64-bit) implement the cascade/parallel formant synthesizer of
D.H. Klatt (1980), "Software for a cascade/parallel formant synthesizer", JASA 67, 971-995. They are built from
the same source and produce the same samples.

## Using it

- Include `klatt/klatt.h`; link `libklatt32.a` or `libklatt64.a` from `build/32` or `build/64`, or load the DLL
  at run time.
- On 32-bit the functions are `__stdcall`, exported under plain names. Python: `ctypes.WinDLL`. See
  `samples/python/klatt.py`.
- The program and the DLL must have the same bitness.

```c
double g[KLATT_NUM_GLOBAL], f[KLATT_NUM_FRAME];
klatt_param_info pi;
for (int i = 0; i < KLATT_NUM_GLOBAL; i++) { KlattParamInfo(KLATT_GLOBAL, i, &pi); g[i] = pi.def; }
for (int i = 0; i < KLATT_NUM_FRAME; i++) { KlattParamInfo(KLATT_FRAME, i, &pi); f[i] = pi.def; }
f[KP_AV] = 60;                                   /* a neutral vowel */

klatt_h h;
if (KlattCreate(g, &h) < 0) return;
short buf[256];                                  /* at least KlattMaxFrameSamples(h) */
for (int k = 0; k < 200; k++) {                  /* 200 frames of 5 ms = 1 s */
	int n = KlattSynthFrame(h, f, buf, 256);
	/* ... use n samples ... */
}
KlattDestroy(h);
```

## Functions

| Function | Returns |
|---|---|
| `KlattVersion()` | `KLATT_VERSION` (10000 = 1.0.0) |
| `KlattParamCount(kind)` | 21 for `KLATT_FRAME`, 18 for `KLATT_GLOBAL` |
| `KlattParamInfo(kind, i, &info)` | name, unit, description, min, max, default |
| `KlattParamIndex(kind, name)` | the index, case-insensitive; `KLATT_ERR_ARG` if unknown |
| `KlattErrorText(code)` | a short English message |
| `KlattCreate(globals, &h)` | a handle; `globals` holds `KLATT_NUM_GLOBAL` values in table order |
| `KlattMaxFrameSamples(h)` | the largest number of samples one frame can produce |
| `KlattSynthFrame(h, frame, out, max)` | samples written; `frame` holds `KLATT_NUM_FRAME` values |
| `KlattOverloadCount(h)` | samples clipped to 16 bits since create or reset |
| `KlattReset(h, seed)` | clears filters, noise, frame clock and the overload count |
| `KlattDestroy(h)` | frees the handle |

- Values outside a parameter's limits are clamped, NaN becomes the minimum, and frequencies are also kept below
  SR/2.
- Globals are fixed for a handle's lifetime; create a new handle to change them.
- A handle must be used by one thread at a time. Separate handles are independent.
- Frame *k* covers samples `floor(k·SR·FRAME_MS/1000)` to `floor((k+1)·SR·FRAME_MS/1000)`, so the total length is
  exact even when a frame is not a whole number of samples (5 ms at 11025 Hz alternates 55 and 56).
- The same inputs and seed always give the same samples.

## Signal path

- **Voicing:** an impulse at the start of each glottal period, through the glottal resonator RGP (FGP, BGP). That
  goes through the glottal antiresonator RGZ (FGZ, BGZ) scaled by AV, and through RGS (FGP, BGS) scaled by AVS.
  AV, AVS and F0 change only at the start of a period.
- **Noise:** a sum of four uniform random numbers, low-pass filtered (`y = x + 0.9·y[n-1]`), halved in the second
  half of each voiced period. AH scales it into aspiration, AF into frication.
- **Cascade** (SW = 0): voicing + aspiration through the nasal pole (FNP, BNP), nasal zero (FNZ, BNZ) and
  formants R1 to R`NFC`.
- **Parallel:** R1 gets the voicing (SW = 1 only); R2-R6 get the first difference of the voicing (SW = 1 only)
  plus frication; the nasal formant (FNP, BNP) gets the voicing difference; AB passes the R2-R6 input directly.
  Signs alternate: +R1 -R2 +R3 -R4 +R5 -R6.
- **Output:** first difference (lip radiation), gain `K_OUT·10^(G0/20)`, rounded and saturated to 16 bits.
- **Amplitudes:** 0 dB is off; otherwise the linear gain is `10^((dB-80)/20)`.

## Calibration

`make calibrate` measures the unscaled levels and prints the constants in `klatt/klatt.c`:

| Constant | Value | Rule |
|---|---|---|
| `K_OUT` | 4.125e5 | neutral vowel, AV = 60, G0 = 47: peak about 16000 |
| `K_NOISE` | 4.071e-4 | AH = 60 alone has the RMS of AV = 60 alone |
| `K_PAR` | 6.0 | SW = 1 with A1-A4 = 60 has the RMS of the cascade vowel |

Re-run it after any change to the signal path and copy the new values into `klatt.c`.

## Parameters

See Help > Parameter reference in the editor, or `klatt/params.c`. Defaults of the globals follow Table I of the
paper, except SR = 11025 Hz and FRAME_MS = 5 ms, which come from IPOX.
